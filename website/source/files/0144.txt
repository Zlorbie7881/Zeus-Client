package com.zlorbie788.zeus.render;

import net.minecraft.client.Minecraft;
import net.minecraft.world.phys.Vec3;

/**
 * Turns points in the world into points on the screen, so boxes, lines and labels can be drawn over everything (through
 * walls too). It uses your eye position, where you look and the field of view setting.
 */
public final class Projector {
    private Projector() {}

    private static double ex, ey, ez;
    private static double fx, fy, fz, rx, ry, rz, ux, uy, uz;
    private static double tanY;
    private static double aspect;
    private static int sw;
    private static int sh;
    private static boolean ready;

    /** Reads the camera for this frame. Call once per frame before projecting. */
    public static boolean prepare(Minecraft mc, float partial, int screenW, int screenH) {
        ready = false;
        if (mc.player == null || screenW <= 0 || screenH <= 0) return false;
        double yawDeg;
        double pitchDeg;
        if (CameraControl.overridingPosition()) {
            double[] e = CameraControl.eye(partial);
            ex = e[0];
            ey = e[1];
            ez = e[2];
        } else {
            Vec3 eye = mc.player.getEyePosition(partial);
            ex = eye.x;
            ey = eye.y;
            ez = eye.z;
        }
        if (CameraControl.overridingLook()) {
            yawDeg = CameraControl.yaw(partial);
            pitchDeg = CameraControl.pitch(partial);
        } else {
            yawDeg = mc.player.getViewYRot(partial);
            pitchDeg = mc.player.getViewXRot(partial);
        }
        double yaw = Math.toRadians(yawDeg);
        double pitch = Math.toRadians(pitchDeg);
        double cp = Math.cos(pitch);
        fx = -Math.sin(yaw) * cp;
        fy = -Math.sin(pitch);
        fz = Math.cos(yaw) * cp;
        rx = -Math.cos(yaw);
        ry = 0;
        rz = -Math.sin(yaw);
        // up = right x forward
        ux = ry * fz - rz * fy;
        uy = rz * fx - rx * fz;
        uz = rx * fy - ry * fx;
        double fov = mc.options.fov().get();
        tanY = Math.tan(Math.toRadians(fov) / 2.0);
        sw = screenW;
        sh = screenH;
        aspect = screenW / (double) screenH;
        ready = true;
        return true;
    }

    public static double eyeX() {
        return ex;
    }

    public static double eyeY() {
        return ey;
    }

    public static double eyeZ() {
        return ez;
    }

    /** Depth of a point in front of the camera (negative: behind). */
    public static double depth(double x, double y, double z) {
        return (x - ex) * fx + (y - ey) * fy + (z - ez) * fz;
    }

    /** Projects a point; fills out[0], out[1] with GUI coordinates and returns false if it is behind you. */
    public static boolean project(double x, double y, double z, float[] out) {
        if (!ready) return false;
        double dx = x - ex;
        double dy = y - ey;
        double dz = z - ez;
        double cz = dx * fx + dy * fy + dz * fz;
        if (cz < 0.05) return false;
        double cx = dx * rx + dy * ry + dz * rz;
        double cy = dx * ux + dy * uy + dz * uz;
        out[0] = (float) (sw / 2.0 + (cx / cz) / (tanY * aspect) * sw / 2.0);
        out[1] = (float) (sh / 2.0 - (cy / cz) / tanY * sh / 2.0);
        return true;
    }

    /**
     * Projects a line between two world points, cut at the near plane so lines that pass behind you still draw.
     * Returns false if nothing is in front.
     */
    public static boolean projectLine(double x1, double y1, double z1, double x2, double y2, double z2, float[] out) {
        double d1 = depth(x1, y1, z1);
        double d2 = depth(x2, y2, z2);
        double near = 0.1;
        if (d1 < near && d2 < near) return false;
        if (d1 < near) {
            double t = (near - d1) / (d2 - d1);
            x1 += (x2 - x1) * t;
            y1 += (y2 - y1) * t;
            z1 += (z2 - z1) * t;
        } else if (d2 < near) {
            double t = (near - d2) / (d1 - d2);
            x2 += (x1 - x2) * t;
            y2 += (y1 - y2) * t;
            z2 += (z1 - z2) * t;
        }
        float[] a = new float[2];
        float[] b = new float[2];
        if (!project(x1, y1, z1, a) || !project(x2, y2, z2, b)) return false;
        out[0] = a[0];
        out[1] = a[1];
        out[2] = b[0];
        out[3] = b[1];
        return true;
    }

    /**
     * Where a world point is on the screen, for a tracer. A point in front of you gives its real place (return 0). A point
     * off to the side, above, below or behind you gives the spot on the edge of the screen it lies towards (return 1), so a
     * line to it still points the right way.
     */
    public static int screenPoint(double x, double y, double z, float margin, float[] out) {
        if (!ready) return -1;
        double dx = x - ex;
        double dy = y - ey;
        double dz = z - ez;
        double cz = dx * fx + dy * fy + dz * fz;
        double cx = dx * rx + dy * ry + dz * rz;
        double cy = dx * ux + dy * uy + dz * uz;
        double px;
        double py;
        if (cz > 0.05) {
            px = sw / 2.0 + (cx / cz) / (tanY * aspect) * sw / 2.0;
            py = sh / 2.0 - (cy / cz) / tanY * sh / 2.0;
            if (px >= margin && px <= sw - margin && py >= margin && py <= sh - margin) {
                out[0] = (float) px;
                out[1] = (float) py;
                return 0;
            }
        } else {
            // behind the camera: the direction on screen is the opposite of where it would project
            px = sw / 2.0 - cx * 1000.0;
            py = sh / 2.0 + cy * 1000.0;
        }
        double vx = px - sw / 2.0;
        double vy = py - sh / 2.0;
        if (Math.abs(vx) < 1e-6 && Math.abs(vy) < 1e-6) vy = 1;
        double hx = sw / 2.0 - margin;
        double hy = sh / 2.0 - margin;
        double k = Math.min(hx / Math.max(1e-6, Math.abs(vx)), hy / Math.max(1e-6, Math.abs(vy)));
        out[0] = (float) (sw / 2.0 + vx * k);
        out[1] = (float) (sh / 2.0 + vy * k);
        return 1;
    }

    public static int width() {
        return sw;
    }

    public static int height() {
        return sh;
    }
}
