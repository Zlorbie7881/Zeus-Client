package com.zlorbie788.zeus.ui;

import java.awt.AlphaComposite;
import java.awt.BasicStroke;
import java.awt.Color;
import java.awt.GradientPaint;
import java.awt.Graphics2D;
import java.awt.RadialGradientPaint;
import java.awt.RenderingHints;
import java.awt.geom.Path2D;
import java.awt.geom.RoundRectangle2D;
import java.awt.image.BufferedImage;
import java.io.InputStream;
import javax.imageio.ImageIO;

/**
 * The picture behind the panel: a deep blue night, soft glows, and branching lightning that crawls along the edges and
 * the top-right corner, drawn once at the real pixel size with Java2D and shown with a single image call per frame.
 * It never touches the middle where the module list sits, so text stays easy to read.
 */
final class PanelArt {
    private PanelArt() {}

    private static BufferedImage referenceHeader;
    private static boolean headerLoaded;

    /** The clean central artwork from the supplied Zeus mockup, feathered so it blends into the generated storm. */
    private static BufferedImage referenceHeader() {
        if (headerLoaded) return referenceHeader;
        headerLoaded = true;
        try (InputStream in = PanelArt.class.getResourceAsStream("/assets/zeus/textures/gui/panel_header.png")) {
            if (in != null) referenceHeader = ImageIO.read(in);
        } catch (Exception ignored) {
            // The procedural artwork below remains a complete fallback if the optional image is unavailable.
        }
        return referenceHeader;
    }

    private static final class Bolt {
        final double x1, y1, x2, y2, wander;
        final int seed;
        final float width;

        Bolt(double x1, double y1, double x2, double y2, double wander, int seed, float width) {
            this.x1 = x1;
            this.y1 = y1;
            this.x2 = x2;
            this.y2 = y2;
            this.wander = wander;
            this.seed = seed;
            this.width = width;
        }
    }

    // positions are fractions of the picture (x right, y down)
    private static final Bolt[] BOLTS = {
        // top right: the big storm over the header
        new Bolt(1.00, 0.00, 0.80, 0.115, 0.10, 11, 1.7f),
        new Bolt(0.99, 0.03, 0.66, 0.05, 0.12, 12, 1.3f),
        new Bolt(0.93, 0.00, 0.60, 0.10, 0.10, 13, 1.1f),
        new Bolt(0.97, 0.00, 0.88, 0.16, 0.12, 14, 1.4f),
        new Bolt(1.00, 0.09, 0.84, 0.025, 0.10, 15, 1.0f),
        new Bolt(0.86, 0.00, 0.50, 0.03, 0.09, 16, 0.9f),
        // top left, over the Z
        new Bolt(0.00, 0.10, 0.12, 0.02, 0.10, 17, 1.0f),
        new Bolt(0.02, 0.00, 0.20, 0.07, 0.10, 18, 0.9f),
        // left edge, running down beside the sidebar
        new Bolt(0.004, 0.20, 0.022, 0.55, 0.05, 19, 1.4f),
        new Bolt(0.012, 0.50, 0.006, 0.86, 0.06, 20, 1.6f),
        new Bolt(0.000, 0.62, 0.05, 0.80, 0.08, 21, 1.0f),
        // bottom left
        new Bolt(0.02, 0.86, 0.01, 1.00, 0.06, 23, 1.0f),
        // right edge
        new Bolt(0.998, 0.20, 0.99, 0.40, 0.04, 24, 0.9f),
        new Bolt(1.00, 0.80, 0.93, 1.00, 0.08, 25, 1.0f),
    };

    static int[] render(int w, int h, int radius) {
        BufferedImage img = new BufferedImage(w, h, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g = img.createGraphics();
        g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g.setRenderingHint(RenderingHints.KEY_STROKE_CONTROL, RenderingHints.VALUE_STROKE_PURE);
        g.setClip(new RoundRectangle2D.Double(0, 0, w, h, radius * 2.0, radius * 2.0));
        double k = Math.max(0.3, w / 1672.0);
        // night
        g.setPaint(new GradientPaint(0, 0, new Color(5, 9, 22), 0, h, new Color(3, 6, 15)));
        g.fillRect(0, 0, w, h);
        // a stormy glow in the middle of the sky behind the statue
        glow(g, w * 0.49, h * 0.10, h * 0.34, 0.34f, 40, 120, 255);
        // glows
        glow(g, w * 0.95, h * 0.02, w * 0.42, 0.50f, 30, 100, 255);
        glow(g, w * 0.02, h * 0.45, h * 0.55, 0.20f, 20, 80, 230);
        glow(g, w * 0.04, h * 1.00, h * 0.40, 0.28f, 25, 90, 255);
        glow(g, w * 0.14, h * 0.07, h * 0.30, 0.16f, 40, 110, 255);
        // the blocky ruins and the statue of Zeus, in the sky of the header
        ruin(g, 0.000, 0.30, new int[] {7, 9, 6, 8, 4, 5}, 1, k, w);
        ruin(g, 0.060, 0.12, new int[] {3, 5, 2}, 1, k, w);
        ruin(g, 0.745, 0.12, new int[] {3, 6, 4, 8, 5}, 1, k, w);
        ruin(g, 0.842, 0.30, new int[] {6, 10, 8, 11, 7, 9, 12}, 1, k, w);
        statue(g, w * 0.485, -h * 0.005, 11.5 * k);
        // Blend in the supplied mockup's central Zeus/ruins artwork. The interactive controls are still drawn by ZeusScreen.
        BufferedImage header = referenceHeader();
        if (header != null) {
            int hx = Math.round(w * 0.3125f);
            int hy = 0;
            int hw = Math.round(w * 0.347f);
            int hh = Math.round(h * 0.251f);
            g.drawImage(header, hx, hy, hx + hw, hy + hh, 0, 0, header.getWidth(), header.getHeight(), null);
        }
        // lightning
        double s = Math.max(0.5, w / 1672.0);
        for (Bolt b : BOLTS) bolt(g, b.x1 * w, b.y1 * h, b.x2 * w, b.y2 * h, b.wander, b.seed, (float) (b.width * s), 2);
        // the working areas sit calmer: the storm shows through, dimmed, behind the sidebar card, the list card and the footer
        g.setComposite(AlphaComposite.SrcOver.derive(0.50f));
        g.setColor(new Color(2, 4, 10));
        g.fillRect((int) (w * 0.236), (int) (h * 0.252), (int) (w * 0.74), (int) (h * 0.675));
        g.setComposite(AlphaComposite.SrcOver.derive(0.40f));
        g.fillRect((int) (w * 0.030), (int) (h * 0.252), (int) (w * 0.196), (int) (h * 0.675));
        g.setComposite(AlphaComposite.SrcOver.derive(0.50f));
        g.fillRect(0, (int) (h * 0.928), w, h);
        g.setComposite(AlphaComposite.SrcOver);
        g.dispose();
        return img.getRGB(0, 0, w, h, null, 0, w);
    }

    private static final String[] HEAD = {
        "...c..c..c.c",
        "..cc.cc.cccc",
        "..cccccccccc",
        "..hhhhhhhhhh",
        ".hhhhhhhhhhh",
        ".hhfffffffff",
        ".hhfffffffff",
        ".hhfdddfffff",
        ".hhfeedfffff",
        ".hhfeefffffd",
        ".hhffffffffd",
        ".hhhfffffffd",
        ".hhbffffbbbb",
        ".hbbbbbbbbbb",
        "hbbbbbbbbbbb",
        "hbbbbbbbbbbb",
        "hhbbbbbbbbbb",
        ".ssbbbbbbbbb",
        "sssssbbbbbbb",
        "sssssssbbbbb",
        "ssssssssssss",
        "ssssssssssss",
    };

    private static Color stone(char c, double lit, double rnd) {
        int r;
        int gg;
        int b;
        switch (c) {
            case 'c' -> { r = 170; gg = 205; b = 255; }
            case 'h' -> { r = 78; gg = 120; b = 205; }
            case 'f' -> { r = 128; gg = 168; b = 240; }
            case 'd' -> { r = 36; gg = 66; b = 140; }
            case 'e' -> { r = 215; gg = 248; b = 255; }
            case 'b' -> { r = 62; gg = 98; b = 188; }
            default -> { r = 40; gg = 68; b = 140; }
        }
        double f = (0.80 + 0.30 * rnd) * lit;
        return new Color(clamp(r * f), clamp(gg * f), clamp(b * f));
    }

    private static int clamp(double v) {
        return (int) Math.max(0, Math.min(255, v));
    }

    /** A pixel-art stone head of Zeus with a crown, a beard and glowing eyes, drawn from the top of the picture. */
    private static void statue(Graphics2D g, double cx, double top, double cell) {
        int cols = HEAD[0].length() * 2;
        double x0 = cx - cols * cell / 2;
        glow(g, cx, top + cell * 8, cell * 16, 0.38f, 60, 140, 255);
        for (int r = 0; r < HEAD.length; r++) {
            for (int c = 0; c < cols; c++) {
                int cc = c < 12 ? c : 23 - c;
                char ch = HEAD[r].charAt(cc);
                if (ch == '.') continue;
                double lit = 1.12 - r * 0.018 + (c < 8 ? 0.12 : c > 15 ? -0.10 : 0.0);
                double rnd = Lightning.hash(r * 31 + c, 7);
                Color col = stone(ch, lit, rnd);
                int x = (int) Math.round(x0 + c * cell);
                int y = (int) Math.round(top + r * cell);
                int sz = (int) Math.ceil(cell);
                g.setColor(col);
                g.fillRect(x, y, sz, sz);
                // a thin lit edge on the top of each block, a dark one on the bottom
                g.setColor(new Color(255, 255, 255, 26));
                g.fillRect(x, y, sz, Math.max(1, sz / 8));
                g.setColor(new Color(0, 10, 40, 60));
                g.fillRect(x, y + sz - Math.max(1, sz / 8), sz, Math.max(1, sz / 8));
                if (ch == 'e') glow(g, x + sz / 2.0, y + sz / 2.0, cell * 2.6, 0.55f, 120, 220, 255);
            }
        }
        // a dark fade at the bottom where the cards start
        g.setPaint(new GradientPaint(0, (float) (top + HEAD.length * cell * 0.62), new Color(3, 6, 15, 0), 0, (float) (top + HEAD.length * cell), new Color(3, 6, 15, 230)));
        g.fillRect((int) (x0 - cell), (int) (top + HEAD.length * cell * 0.6), (int) (cols * cell + 2 * cell), (int) (HEAD.length * cell * 0.45));
    }

    /** A stepped silhouette of broken stone blocks, standing on the left or right of the header. */
    private static void ruin(Graphics2D g, double xFrac, double hFrac, int[] heights, int dir, double k, int w) {
        double cw = 24 * k;
        double ch = 22 * k;
        double x0 = xFrac * w;
        for (int i = 0; i < heights.length; i++) {
            for (int j = 0; j < heights[i]; j++) {
                double rnd = Lightning.hash(i * 17 + j, 3 + (int) (xFrac * 100));
                if (j == heights[i] - 1 && rnd < 0.4) continue; // a broken top
                int x = (int) Math.round(x0 + i * cw);
                int y = (int) Math.round(j * ch);
                int sz = (int) Math.ceil(cw);
                int sh = (int) Math.ceil(ch);
                double lit = 0.75 + 0.5 * rnd - j * 0.012;
                g.setColor(new Color(clamp(18 * lit + 8), clamp(34 * lit + 14), clamp(78 * lit + 30)));
                g.fillRect(x, y, sz, sh);
                g.setColor(new Color(70, 130, 255, 34));
                g.fillRect(x, y, sz, Math.max(1, sh / 8));
                g.setColor(new Color(0, 6, 24, 90));
                g.fillRect(x, y + sh - Math.max(1, sh / 10), sz, Math.max(1, sh / 10));
                g.fillRect(x + sz - Math.max(1, sz / 12), y, Math.max(1, sz / 12), sh);
            }
        }
    }

    private static void glow(Graphics2D g, double cx, double cy, double r, float strength, int rr, int gg, int bb) {
        float[] fr = {0f, 0.55f, 1f};
        Color[] cols = {new Color(rr, gg, bb, (int) (255 * strength)), new Color(rr, gg, bb, (int) (255 * strength * 0.22f)), new Color(rr, gg, bb, 0)};
        g.setPaint(new RadialGradientPaint((float) cx, (float) cy, (float) r, fr, cols));
        g.fillRect((int) (cx - r), (int) (cy - r), (int) (2 * r), (int) (2 * r));
    }

    private static void bolt(Graphics2D g, double x1, double y1, double x2, double y2, double wander, int seed, float width, int depth) {
        double len = Math.hypot(x2 - x1, y2 - y1);
        if (len < 6) return;
        int levels = Math.max(3, Math.min(7, (int) Math.round(Math.log(len / 5.0) / Math.log(2))));
        float[][] p = Lightning.path((float) x1, (float) y1, (float) x2, (float) y2, seed, (float) (len * wander), levels, 0.62f);
        int n = p[0].length;
        Path2D.Double path = new Path2D.Double();
        path.moveTo(p[0][0], p[1][0]);
        for (int i = 1; i < n; i++) path.lineTo(p[0][i], p[1][i]);
        stroke(g, path, width);
        if (depth <= 0) return;
        double ux = (x2 - x1) / len, uy = (y2 - y1) / len;
        for (int k = 0; k < 3; k++) {
            int at = 2 + (int) (Lightning.hash(seed, 100 + k) * Math.max(1, n - 5));
            if (at >= n - 1) continue;
            double side = Lightning.hash(seed, 110 + k) < 0.5 ? -1 : 1;
            double ang = side * (0.35 + 0.7 * Lightning.hash(seed, 120 + k));
            double ca = Math.cos(ang), sa = Math.sin(ang);
            double bdx = ux * ca - uy * sa, bdy = ux * sa + uy * ca;
            double bl = len * (0.15 + 0.22 * Lightning.hash(seed, 130 + k));
            double bx = p[0][at], by = p[1][at];
            bolt(g, bx, by, bx + bdx * bl, by + bdy * bl, wander, seed * 31 + k * 17 + 5, width * 0.6f, depth - 1);
        }
    }

    private static void stroke(Graphics2D g, Path2D.Double path, float width) {
        g.setStroke(new BasicStroke(width * 9f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g.setColor(new Color(40, 110, 255, 22));
        g.draw(path);
        g.setStroke(new BasicStroke(width * 4.2f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g.setColor(new Color(60, 140, 255, 70));
        g.draw(path);
        g.setStroke(new BasicStroke(width * 1.9f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g.setColor(new Color(110, 180, 255, 190));
        g.draw(path);
        g.setStroke(new BasicStroke(Math.max(1f, width * 0.8f), BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g.setColor(new Color(225, 240, 255, 245));
        g.draw(path);
    }
}
