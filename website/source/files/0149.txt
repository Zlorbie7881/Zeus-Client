 * colour passed when drawing, so one drawing serves every colour and every state.

    // ------------------------------------------------------------------ small drawing helpers (unit space 0..100)

    private static void ln(Graphics2D g, double x1, double y1, double x2, double y2) {
        Path2D.Double p = new Path2D.Double();
        p.moveTo(x1, y1);
        p.lineTo(x2, y2);
        g.draw(p);
    }

    private static Path2D.Double path(double... pts) {
        Path2D.Double p = new Path2D.Double();
        p.moveTo(pts[0], pts[1]);
        for (int i = 2; i < pts.length; i += 2) p.lineTo(pts[i], pts[i + 1]);
        return p;
    }

    private static void poly(Graphics2D g, boolean close, double... pts) {
        Path2D.Double p = path(pts);
        if (close) p.closePath();
        g.draw(p);
    }

    private static void fillPoly(Graphics2D g, double... pts) {
        Path2D.Double p = path(pts);
        p.closePath();
        g.fill(p);
    }

    /** The ten corners of a five pointed star, as x, y pairs. */
    private static double[] starPoints(double cx, double cy, double outer, double inner) {
        double[] p = new double[20];
        for (int i = 0; i < 10; i++) {
            double a = -Math.PI / 2 + i * Math.PI / 5;
            double r = i % 2 == 0 ? outer : inner;
            p[i * 2] = cx + Math.cos(a) * r;
            p[i * 2 + 1] = cy + Math.sin(a) * r;
        }
        return p;
    }

    private static void ring(Graphics2D g, double cx, double cy, double r) {
        g.draw(new Ellipse2D.Double(cx - r, cy - r, 2 * r, 2 * r));
    }

    private static void disc(Graphics2D g, double cx, double cy, double r) {
        g.fill(new Ellipse2D.Double(cx - r, cy - r, 2 * r, 2 * r));
    }

    private static void arc(Graphics2D g, double cx, double cy, double r, double start, double extent) {
        g.draw(new Arc2D.Double(cx - r, cy - r, 2 * r, 2 * r, start, extent, Arc2D.OPEN));
    }

    private static void box(Graphics2D g, double x, double y, double w, double h, double r) {
        g.draw(new RoundRectangle2D.Double(x, y, w, h, r, r));
    }

    private static void sword(Graphics2D g, double ax, double ay, double bx, double by) {
        // a blade from the hilt (a) to the tip (b), a cross guard and a pommel
        double dx = bx - ax, dy = by - ay;
        double len = Math.hypot(dx, dy);
        double ux = dx / len, uy = dy / len, nx = -uy, ny = ux;
        double gx = ax + ux * len * 0.28, gy = ay + uy * len * 0.28;
        ln(g, ax, ay, bx, by);
        ln(g, gx - nx * 13, gy - ny * 13, gx + nx * 13, gy + ny * 13);
        disc(g, ax - ux * 4, ay - uy * 4, 4.2);
    }

    // ------------------------------------------------------------------ the atlas

    private static Tex.Img atlas(int cell) {
        Tex.Img have = ATLAS.get(cell);
        if (have != null) return have;
        int rows = (NAMES.length + COLS - 1) / COLS;
        int w = COLS * cell;
        int h = rows * cell;
        int[] argb = new int[w * h];
        for (int i = 0; i < NAMES.length; i++) {
            int[] px = render(NAMES[i], cell);
            int ox = (i % COLS) * cell;
            int oy = (i / COLS) * cell;
            for (int y = 0; y < cell; y++) System.arraycopy(px, y * cell, argb, (oy + y) * w + ox, cell);
        }
        Tex.Img img = Tex.made("glyphs" + cell, argb, w, h);
        ATLAS.put(cell, img);
        return img;
    }

    /** One drawing at {@code cell} pixels square: a crisp white line drawing over a soft white halo. */
    static int[] render(String name, int cell) {
        int pad = Math.max(2, cell / 12);
        BufferedImage line = new BufferedImage(cell, cell, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g = line.createGraphics();
        g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g.setRenderingHint(RenderingHints.KEY_STROKE_CONTROL, RenderingHints.VALUE_STROKE_PURE);
        double inner = cell - 2.0 * pad;
        g.translate(pad, pad);
        g.scale(inner / 100.0, inner / 100.0);
        g.setColor(java.awt.Color.WHITE);
        g.setStroke(new BasicStroke(7.2f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        Consumer<Graphics2D> p = PAINT.get(name);
        if (p != null) p.accept(g);
        g.dispose();
        int[] a = line.getRGB(0, 0, cell, cell, null, 0, cell);
        float[] alpha = new float[cell * cell];
        for (int i = 0; i < alpha.length; i++) alpha[i] = (a[i] >>> 24) / 255f;
        float[] halo = blur(alpha, cell, cell, Math.max(1, cell / 16));
        int[] out = new int[cell * cell];
        for (int i = 0; i < out.length; i++) {
            float core = alpha[i];
            float soft = Math.min(1f, halo[i] * 1.5f) * 0.45f;
            float v = core + (1f - core) * soft;
            out[i] = ((int) (Math.min(1f, v) * 255) << 24) | 0xFFFFFF;
        }
        return out;
    }

    private static float[] blur(float[] src, int w, int h, int r) {
        float[] cur = src;
        for (int pass = 0; pass < 2; pass++) {
            float[] tmp = new float[w * h];
            for (int y = 0; y < h; y++) {
                float sum = 0;
                for (int x = -r; x <= r; x++) sum += cur[y * w + Math.max(0, Math.min(w - 1, x))];
                for (int x = 0; x < w; x++) {
                    tmp[y * w + x] = sum / (2 * r + 1);
                    sum += cur[y * w + Math.min(w - 1, x + r + 1)] - cur[y * w + Math.max(0, x - r)];
                }
            }
            float[] out = new float[w * h];
            for (int x = 0; x < w; x++) {
                float sum = 0;
                for (int y = -r; y <= r; y++) sum += tmp[Math.max(0, Math.min(h - 1, y)) * w + x];
                for (int y = 0; y < h; y++) {
                    out[y * w + x] = sum / (2 * r + 1);
                    sum += tmp[Math.min(h - 1, y + r + 1) * w + x] - tmp[Math.max(0, y - r) * w + x];
                }
            }
            cur = out;
        }
        return cur;
    }

    /** Draws icon {@code name}, centred on (cx, cy), {@code size} interface pixels square, tinted. */
    public static void draw(GuiGraphics g, String name, float cx, float cy, float size, int tint) {
        Integer idx = INDEX.get(name);
        if (idx == null) return;
        int gs = Tex.guiScale();
        int cell = Math.max(16, Math.round(size * gs));
        Tex.Img img = atlas(cell);
        int sx = (idx % COLS) * cell;
        int sy = (idx / COLS) * cell;
        int dw = Math.round(size);
        int x = Math.round(cx - size / 2f);
        int y = Math.round(cy - size / 2f);
        img.drawPart(g, x, y, dw, dw, sx, sy, cell, cell, tint);
    }

    public static void releaseAll() {
        ATLAS.clear();
    }
