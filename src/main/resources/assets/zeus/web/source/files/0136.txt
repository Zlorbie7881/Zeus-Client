/**
 * Zeus Sense and Threat Radar: the two modules that look at the whole situation and tell you what matters, most urgent first.
 * Both use {@link #scan}, one shared look at the players around you, so neither costs more when both are on.
 */
public final class Sense {
    private Sense() {}

    /** A player rated by {@link Threat}. */
    public record Rated(UUID id, String name, Threat.Result result, double dist, String gear, boolean spectator) {}

    private static final Map<UUID, double[]> LAST = new HashMap<>();
    private static List<Rated> cache = List.of();
    private static long cacheAt;
    private static int cacheRange;

    private static final int[] LEVEL_COLORS = {0xFF5EE6A0, 0xFFFFC857, 0xFFFFA24D, 0xFFFF4D5E};

    /** Rates every other player within {@code range} blocks. Cached for a fraction of a second. Strongest threat first. */
    public static List<Rated> scan(Minecraft mc, int range) {
        long now = System.nanoTime();
        if (now - cacheAt < 120_000_000L && range <= cacheRange) return cache;
        if (mc.player == null || mc.level == null) return List.of();
        List<Rated> out = new ArrayList<>();
        Map<UUID, double[]> seen = new HashMap<>();
        Player me = mc.player;
        for (var p : mc.level.players()) {
            if (p == me || !p.isAlive()) continue;
            double dist = Math.sqrt(me.distanceToSqr(p));
            if (dist > range) continue;
            double[] prev = LAST.get(p.getUUID());
            double closing = 0;
            if (prev != null) {
                double dt = (now - (long) prev[1]) / 1e9;
                if (dt > 0.05) closing = prev[2] * 0.5 + 0.5 * (prev[0] - dist) / dt;
                else closing = prev[2];
            }
            seen.put(p.getUUID(), new double[] {dist, now, closing});
            int armor = 0;
            boolean elytra = false;
            for (EquipmentSlot s : new EquipmentSlot[] {EquipmentSlot.HEAD, EquipmentSlot.CHEST, EquipmentSlot.LEGS, EquipmentSlot.FEET}) {
                ItemStack st = p.getItemBySlot(s);
                armor += Gear.armorRank(st);
                if (s == EquipmentSlot.CHEST && Names.item(st).equals("elytra")) elytra = true;
            }
            String main = Names.item(p.getMainHandItem());
            String off = Names.item(p.getOffhandItem());
            var look = p.getLookAngle();
            double dx = me.getX() - p.getX();
            double dy = me.getY() + 1 - (p.getY() + 1.6);
            double dz = me.getZ() - p.getZ();
            double len = Math.sqrt(dx * dx + dy * dy + dz * dz);
            boolean aimed = len > 0.1 && (look.x * dx + look.y * dy + look.z * dz) / len > 0.97;
            Threat.Result r = Threat.rate(new Threat.Input(armor, main, off, dist, closing, aimed, p.getHealth(), p.getMaxHealth(), p.isInvisible(), elytra));
            out.add(new Rated(p.getUUID(), p.getName().getString(), r, dist, gearWord(armor, main, off), p.isSpectator()));
        }
        LAST.clear();
        LAST.putAll(seen);
        out.sort((a, b) -> Integer.compare(b.result.score(), a.result.score()));
        cache = out;
        cacheAt = now;
        cacheRange = range;
        return out;
    }

    static String gearWord(int armor, String main, String off) {
        String a = armor >= 240 ? "netherite" : armor >= 170 ? "diamond+" : armor >= 100 ? "iron" : armor > 0 ? "light" : "no armor";
        String w = main.isEmpty() || main.equals("air") ? "" : main.replace('_', ' ');
        return w.isEmpty() ? a : a + ", " + w;
    }

    // ------------------------------------------------------------------ Threat Radar

    /** Lists the players around you, the most dangerous first, with how dangerous and why. */
    public static final class ThreatRadar extends Module {
        private final Setting.Choice corner = Hud.corner(this, 3);
        private final Setting.Num range = add(new Setting.Num("Range", "How far away players are rated.", 64, 16, 160, 8, " blocks"));
        private final Setting.Num most = add(new Setting.Num("Most rows", "The most dangerous are kept.", 5, 2, 10, 1, ""));
        private final Setting.Choice from = add(new Setting.Choice("Show from", "Players rated below this are not listed.", 0, Threat.LEVELS));
        private final Setting.Bool reasons = add(new Setting.Bool("Show reasons", "A line under each name that says why it is rated like that.", true));
        private final Setting.Bool alert = add(new Setting.Bool("Warn when someone becomes a danger", "One message when a player first reaches Danger.", true));
        private final Setting.Bool spectators = add(new Setting.Bool("Include spectators", "Spectators cannot fight you, so they are left out unless this is on.", false));
        private final Setting.Num scale = add(new Setting.Num("Size", "The size of the box.", 100, 70, 160, 5, " %"));
        private final Setting.Num opacity = add(new Setting.Num("Background", "How solid the box is.", 88, 30, 100, 5, " %"));
        private final Map<UUID, Long> warned = new HashMap<>();
        private List<Rated> shown = List.of();

        public ThreatRadar() {
            super("threatradar", "Threat Radar", "Rates the players around you by how dangerous they are, from their gear, weapon, distance and aim.", Category.PANELS);
            anywhere();
        }

        @Override
        protected void onEnable() {
            warned.clear();
            shown = List.of();
        }

        @Override
        protected void onTick(Minecraft mc) {
            List<Rated> all = scan(mc, range.i());
            List<Rated> out = new ArrayList<>();
            for (Rated r : all) {
                if (r.spectator && !spectators.on()) continue;
                if (r.result.score() < from.index() * 25) continue;
                out.add(r);
                if (alert.on() && r.result.score() >= 75) {
                    long now = System.currentTimeMillis();
                    Long last = warned.get(r.id);
                    if (last == null || now - last > 45_000) {
                        warned.put(r.id, now);
                        Zeus.toast("Danger: " + r.name + " (" + r.result.score() + ", " + Math.round(r.dist) + "m): " + String.join(", ", r.result.reasons().stream().limit(2).toList()), false);
                    }
                }
            }
            shown = out;
        }

        @Override
        public String info() {
            return shown.isEmpty() ? null : Integer.toString(shown.get(0).result.score());
        }

        @Override
        protected void onHud(GuiGraphics g, Minecraft mc, float partial, int w, int h) {
            List<Rated> list = shown;
            if (list.isEmpty()) return;
            float sc = scale.f() / 100f;
            int n = Math.min(list.size(), most.i());
            boolean why = reasons.on();
            int rh = Math.round((why ? 24 : 15) * sc);
            String title = "Threats · " + list.size();
            int bw = Draw.widthBoldS(title, sc) + Math.round(24 * sc);
            for (int i = 0; i < n; i++) {
                Rated r = list.get(i);
                int rw = Draw.widthS(r.name, sc) + Draw.widthS(r.result.level() + " " + r.result.score() + "  " + Math.round(r.dist) + "m", sc) + Math.round(34 * sc);
                if (why) rw = Math.max(rw, Math.min(Draw.widthS(r.gear + " · " + String.join(", ", r.result.reasons()), sc * 0.8f) + Math.round(16 * sc), Math.round(260 * sc)));
                bw = Math.max(bw, rw);
            }
            bw = Math.min(bw, Math.round(280 * sc));
            int head = Math.round(20 * sc);
            int bh = head + n * rh + 4;
            int[] at = Hud.place(corner.index(), bw, bh, w, h);
            float bg = opacity.f() / 100f;
            int top = LEVEL_COLORS[indexOf(list.get(0).result.level())];
            Draw.shadow(g, at[0], at[1], bw, bh, 7, 5, 0.35f * bg);
            Draw.rrect(g, at[0], at[1], bw, bh, 7, Draw.rgba(0xFF0D1220, bg), Draw.rgba(0xFF121829, bg));
            Draw.rrectLine(g, at[0], at[1], bw, bh, 7, Draw.rgba(top, 0.6f));
            Draw.boldS(g, title, at[0] + Math.round(8 * sc), at[1] + Math.round(6 * sc), 0xFFFFFFFF, sc);
            g.fill(at[0] + 8, at[1] + head - 2, at[0] + bw - 8, at[1] + head - 1, Draw.rgba(top, 0.4f));
            int y = at[1] + head + 1;
            for (int i = 0; i < n; i++) {
                Rated r = list.get(i);
                int col = LEVEL_COLORS[indexOf(r.result.level())];
                int x = at[0] + Math.round(8 * sc);
                int ty = y + Math.round(2 * sc);
                // a bar that shows the score
                g.fill(x - 4, y + 1, x - 2, y + rh - Math.round(3 * sc), Draw.rgba(col, 0.9f));
                Draw.textS(g, r.name, x, ty, 0xFFFFFFFF, sc);
                String right = r.result.level() + " " + r.result.score() + "  " + Math.round(r.dist) + "m";
                Draw.textRightS(g, right, at[0] + bw - Math.round(8 * sc), ty, col, sc);
                if (why) {
                    String line = r.gear + (r.result.reasons().isEmpty() ? "" : " · " + String.join(", ", r.result.reasons()));
                    int max = bw - Math.round(16 * sc);
                    String t = line;
                    while (t.length() > 4 && Draw.widthS(t, sc * 0.8f) > max) t = t.substring(0, t.length() - 2);
                    Draw.textS(g, t.equals(line) ? t : t + "…", x, ty + Math.round(10 * sc), 0xFF8B97B3, sc * 0.8f);
                }
                y += rh;
            }
        }

        private static int indexOf(String level) {
            for (int i = 0; i < Threat.LEVELS.length; i++) if (Threat.LEVELS[i].equals(level)) return i;
            return 0;
        }
    }

    // ------------------------------------------------------------------ Zeus Sense

    private record Alert(String id, int pri, String text) {}

    private static final int[] PRI_COLORS = {0xFFFF4D5E, 0xFFFFA24D, 0xFFFFC857, 0xFF8FB4FF};

    /**
     * Zeus Sense watches your own state and your surroundings and keeps a short list of what needs attention, most urgent first:
     * low health with no totem, a dangerous player coming, a spectator next to you, armor about to break, falling into the
     * void, a raided base in front of you. It reads only what your game already knows.
     */
    public static final class ZeusSense extends Module {
        private final Setting.Choice corner = Hud.corner(this, 0);
        private final Setting.Bool body = add(new Setting.Bool("Health, food and armor", "Low health (and whether you have a totem), hunger, armor about to break, a nearly full inventory.", true));
        private final Setting.Bool players = add(new Setting.Bool("Players around", "Dangerous players near you, spectators close by, no totem when someone is near.", true));
        private final Setting.Bool danger = add(new Setting.Bool("Falling, fire and lava", "Falling into the void, standing in lava or burning.", true));
        private final Setting.Bool bases = add(new Setting.Bool("Bases you found", "When you come near a base Zeus flagged: what it is and whether it looks raided.", true));
        private final Setting.Bool game = add(new Setting.Bool("The game itself", "When your game lags and Zeus slows the scanners down.", true));
        private final Setting.Num lowHealth = add(new Setting.Num("Low health at", "Hearts. Below this it warns.", 3, 1, 9, 0.5, " hearts"));
        private final Setting.Num near = add(new Setting.Num("Player range", "How far away players count as near.", 40, 16, 100, 4, " blocks"));
        private final Setting.Bool toasts = add(new Setting.Bool("Pop-up for urgent ones", "A short message when something Urgent or Important appears (once every 30 seconds for the same thing).", true));
        private final Setting.Bool box = add(new Setting.Bool("Show the list", "The box on the screen. Off keeps only the pop-ups.", true));
        private final Setting.Num scale = add(new Setting.Num("Size", "The size of the box.", 100, 70, 160, 5, " %")).showIf(() -> this.box.on());
        private final Setting.Num opacity = add(new Setting.Num("Background", "How solid the box is.", 88, 30, 100, 5, " %")).showIf(() -> this.box.on());
        private List<Alert> active = List.of();
        private final Map<String, Long> told = new HashMap<>();
        private int ticks;

        public ZeusSense() {
            super("sense", "Zeus Sense", "Watches your health, gear, surroundings and the players around you and lists what needs attention, most urgent first.", Category.UTILITY);
            hudCard();
            anywhere();
            onByDefault();
        }

        @Override
        protected void onEnable() {
            told.clear();
            active = List.of();
        }

        private static boolean totemReady(Minecraft mc) {
            if (Names.item(mc.player.getOffhandItem()).equals("totem_of_undying")) return true;
            for (int i = 0; i < 9; i++) if (Names.item(Inv.get(i)).equals("totem_of_undying")) return true;
            return false;
        }

        @Override
        protected void onTick(Minecraft mc) {
            if (mc.player == null || mc.level == null) return;
            if (++ticks % 8 != 0) return;
            List<Alert> out = new ArrayList<>();
            var me = mc.player;
            Brain.Ctx ctx = Brain.ctx();
            if (body.on()) {
                float hp = me.getHealth();
                if (hp <= lowHealth.f() * 2) {
                    boolean totem = totemReady(mc);
                    out.add(new Alert("hp", 0, String.format(Locale.ROOT, "Health %.1f hearts%s", hp / 2, totem ? "" : ", no totem")));
                }
                int food = me.getFoodData().getFoodLevel();
                if (food <= 6) out.add(new Alert("food", food <= 2 ? 1 : 2, "Hungry: " + food + " / 20"));
                for (EquipmentSlot s : new EquipmentSlot[] {EquipmentSlot.HEAD, EquipmentSlot.CHEST, EquipmentSlot.LEGS, EquipmentSlot.FEET}) {
                    ItemStack st = me.getItemBySlot(s);
                    if (st.isEmpty() || !st.isDamageableItem()) continue;
                    int left = st.getMaxDamage() - st.getDamageValue();
                    if (left * 100 < st.getMaxDamage() * 12) out.add(new Alert("armor" + s.name(), 1, "Almost broken: " + Names.item(st).replace('_', ' ') + " (" + left + " left)"));
                }
                int free = 0;
                for (int i = 0; i < 36; i++) if (Inv.get(i).isEmpty()) free++;
                if (free <= 1) out.add(new Alert("inv", 3, free == 0 ? "Inventory is full" : "Inventory almost full"));
            }
            if (players.on()) {
                List<Rated> rated = scan(mc, near.i());
                boolean someoneNear = false;
                for (Rated r : rated) {
                    if (r.spectator) {
                        if (r.dist <= 48) out.add(new Alert("spec" + r.name, 1, r.name + " is watching you in spectator mode (" + Math.round(r.dist) + "m)"));
                        continue;
                    }
                    int sc = r.result.score();
                    if (sc >= 50) someoneNear = true;
                    if (sc >= 75) out.add(new Alert("thr" + r.name, 0, "Danger: " + r.name + " " + sc + ", " + Math.round(r.dist) + "m"));
                    else if (sc >= 50) out.add(new Alert("thr" + r.name, 2, "Watch out: " + r.name + " " + sc + ", " + Math.round(r.dist) + "m"));
                }
                if (someoneNear && !Names.item(me.getOffhandItem()).equals("totem_of_undying")) out.add(new Alert("nototem", 1, "No totem in your offhand, and someone dangerous is near"));
            }
            if (danger.on()) {
                if (me.isInLava()) out.add(new Alert("lava", 0, "You are in lava"));
                else if (me.isOnFire()) out.add(new Alert("fire", 1, "You are on fire"));
                boolean voidish = ctx.dim.equals("overworld") ? me.getY() < -40 : ctx.dim.equals("end") && me.getY() < 20;
                if (voidish && me.fallDistance > 8 && !me.onGround() && !me.isFallFlying()) out.add(new Alert("void", 0, "Falling into the void"));
            }
            if (bases.on()) {
                Ledger.Entry best = null;
                double bd = 110;
                for (Ledger.Entry e : Ledger.list(ctx.server)) {
                    double d = Math.hypot(e.x - me.getX(), e.z - me.getZ());
                    if (d < bd) {
                        bd = d;
                        best = e;
                    }
                }
                if (best != null) {
                    boolean bad = best.status.startsWith("Raided") || best.status.startsWith("Looted");
                    out.add(new Alert("base", bad ? 1 : 3, (bad ? best.status + ": " : "Near a ") + best.type.toLowerCase(Locale.ROOT) + " you flagged (" + best.status.toLowerCase(Locale.ROOT) + ", " + Math.round(bd) + "m)"));
                }
            }
            if (game.on() && ctx.fps < 25) out.add(new Alert("lag", 3, "Game is lagging (" + ctx.fps + " fps): scanners are slowed down"));
            out.sort((a, b) -> Integer.compare(a.pri, b.pri));
            long now = System.currentTimeMillis();
            if (toasts.on()) {
                for (Alert a : out) {
                    if (a.pri > 1) continue;
                    Long last = told.get(a.id);
                    if (last == null || now - last > 30_000) {
                        told.put(a.id, now);
                        Zeus.toast(a.text, false);
                    }
                }
            }
            active = out;
        }

        @Override
        public String info() {
            return active.isEmpty() ? null : Integer.toString(active.size());
        }

        @Override
        protected void onHud(GuiGraphics g, Minecraft mc, float partial, int w, int h) {
            List<Alert> list = active;
            if (!box.on() || list.isEmpty()) return;
            float sc = scale.f() / 100f;
            int n = Math.min(list.size(), 7);
            int rh = Math.round(13 * sc);
            String title = "Zeus Sense";
            int bw = Draw.widthBoldS(title, sc) + Math.round(24 * sc);
            for (int i = 0; i < n; i++) bw = Math.max(bw, Draw.widthS(list.get(i).text, sc) + Math.round(26 * sc));
            bw = Math.min(bw, Math.round(300 * sc));
            int head = Math.round(19 * sc);
            int bh = head + n * rh + 4;
            int[] at = Hud.place(corner.index(), bw, bh, w, h);
            float bg = opacity.f() / 100f;
            int top = PRI_COLORS[list.get(0).pri];
            Draw.shadow(g, at[0], at[1], bw, bh, 7, 5, 0.35f * bg);
            Draw.rrect(g, at[0], at[1], bw, bh, 7, Draw.rgba(0xFF0D1220, bg), Draw.rgba(0xFF121829, bg));
            Draw.rrectLine(g, at[0], at[1], bw, bh, 7, Draw.rgba(top, 0.6f));
            Draw.boldS(g, title, at[0] + Math.round(8 * sc), at[1] + Math.round(5 * sc), 0xFFFFFFFF, sc);
            int y = at[1] + head;
            for (int i = 0; i < n; i++) {
                Alert a = list.get(i);
                int col = PRI_COLORS[a.pri];
                int ty = y + Math.round(2 * sc);
                g.fill(at[0] + 8, y + 3, at[0] + 10, y + rh - 2, Draw.rgba(col, 0.95f));
                String t = a.text;
                int max = bw - Math.round(26 * sc);
                while (t.length() > 4 && Draw.widthS(t, sc) > max) t = t.substring(0, t.length() - 2);
                Draw.textS(g, t.equals(a.text) ? t : t + "…", at[0] + Math.round(15 * sc), ty, col, sc);
                y += rh;
            }
        }
    }
}
