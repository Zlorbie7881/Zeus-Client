import io
import json

WEB = "src/main/resources/assets/zeus/web/features.json"
WCP = "website/features.json"


def rd(p):
    return io.open(p, encoding="utf-8").read()


def wr(p, s):
    io.open(p, "w", encoding="utf-8", newline="").write(s)


data = json.loads(rd(WEB))
mods = data["modules"]
ids = {m["id"]: m for m in mods}


def switch(name, desc, default):
    return {"name": name, "desc": desc, "type": "switch", "default": default}


def slider(name, desc, lo, hi, step, default, unit=None):
    s = {"name": name, "desc": desc,
         "type": "slider", "min": float(lo), "max": float(hi),
         "step": float(step), "default": float(default)}
    if unit:
        s["unit"] = unit
    return s


def choice(name, desc, options, default):
    return {"name": name, "desc": desc, "type": "choice",
            "options": options, "default": default}


def text(name, desc, default):
    return {"name": name, "desc": desc, "type": "text", "default": default}

#    Rename it to "lagmonitor" so the real module can own "fakelag".
#    If the new Fakelag module is already present, do nothing (idempotent).

elif old.get("name") == "Fake Lag":
    # already the new module; nothing to do
    pass
else:
    raise SystemExit("unexpected fakelag card: %r" % old.get("name"))
