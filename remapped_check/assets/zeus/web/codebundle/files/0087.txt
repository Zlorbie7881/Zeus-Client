package com.zlorbie788.zeus.mods;

import com.zlorbie788.zeus.core.Category;
import com.zlorbie788.zeus.core.Module;
import com.zlorbie788.zeus.core.Setting;
import com.zlorbie788.zeus.render.Projector;
import com.zlorbie788.zeus.ui.AuroraSky;
import com.zlorbie788.zeus.ui.Perf;
import com.zlorbie788.zeus.ui.SkyScape;
import net.minecraft.client.Minecraft;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.core.BlockPos;
import net.minecraft.world.level.Level;
import net.minecraft.world.level.LightLayer;

import java.util.List;

/**
 * A night sky made from a real picture (a Milky Way, big northern lights and dark clouds), laid over the dome of the sky and
 * moved with the camera, and brought to life: the lights sway and flow, the clouds drift, the stars twinkle and new shooting
 * stars fall at random moments (see {@link SkyPhoto}).
 *
 * <p>The picture was taken apart ahead of time (sky without stars, two star layers, a glow layer and a map of where the aurora
 * and the clouds are), so the game only draws a few hundred textured rectangles and does no per-pixel work. Like the Aurora
 * module it is drawn over the game picture from the camera's own view and cut along the skyline of the land, so it goes behind
 * hills and trees; it fades in at night, out at dawn, in rain and under a roof, and Zeus's automatic lightening lowers the
 * detail when the frame rate drops.</p>
 */
public final class Atmosphere extends Module {
    private final Setting.Num brightness = add(new Setting.Num("Brightness", "How strongly the sky covers the vanilla one.", 100, 30, 100, 5, "%"));
    private final Setting.Num motion = add(new Setting.Num("Aurora motion", "How much the northern lights sway and flow and the clouds drift. 0 keeps the picture still.", 100, 0, 250, 10, "%"));
    private final Setting.Bool twinkle = add(new Setting.Bool("Twinkling stars", "The stars of the picture twinkle.", true));
    private final Setting.Choice shooting = add(new Setting.Choice("Shooting stars", "How often a new one falls across the sky.", 2, "Off", "Rare", "Sometimes", "Often"));
    private final Setting.Choice detail = add(new Setting.Choice("Detail", "How finely the picture is cut up to move. Zeus lowers it by itself when the frame rate is low.", 1, "Low", "Medium", "High"));
    private final Setting.Choice when = add(new Setting.Choice("Show", "When the sky is shown.", 0, "Clear nights", "Nights, also in rain", "Always"));
    private final Setting.Bool hills = add(new Setting.Bool("Hide behind land", "The sky goes behind hills, mountains and trees instead of floating in front of them. Switch it off for the cheapest drawing.", true));
    private final Setting.Info status = add(new Setting.Info("Status", this::status));

    private static final float[] METEOR_GAP = {0f, 24f, 11f, 5f};
    private static final int[][] CELLS = {{12, 7}, {19, 10}, {28, 15}};

    private final SkyScape scape = new SkyScape();
    private final com.zlorbie788.zeus.ui.SkyPhoto photo = new com.zlorbie788.zeus.ui.SkyPhoto();
    private final com.zlorbie788.zeus.ui.SkyPhoto.Look look = new com.zlorbie788.zeus.ui.SkyPhoto.Look();
    private final Aurora.Skyline skyline = new Aurora.Skyline();
    private final double[] dir = new double[3];

    private float fade;
    private float clock;
    private long last;
    private String why = "";
    private boolean warned;

    public Atmosphere() {
        super("atmosphere", "Atmosphere", "A night sky made from a real picture: Milky Way, northern lights and dark clouds that move, stars that twinkle and shooting stars that fall.", Category.RENDER);
    }

    @Override
    protected void onDisable() {
        fade = 0f;
        last = 0L;
        skyline.reset();
    }

    @Override
    public String info() {
        return "Night sky";
    }

    private List<String> status() {
        if (!enabled()) return List.of("Switched off");
        return List.of(
                "Showing: " + (fade > 0.02f ? Math.round(fade * 100f) + "%" : "hidden"),
                "Why: " + (why.isEmpty() ? "-" : why),
                "Picture: " + (photo.failed() ? "missing" : photo.ready() ? "loaded" : "loading"),
                "Shooting stars in the sky: " + scape.meteorCount());
    }

    @Override
    protected void onHudWorld(GuiGraphics g, Minecraft mc, float partial, int w, int h) {
        long now = System.nanoTime();
        float dt = last == 0L ? 0f : Math.min(0.1f, (now - last) / 1e9f);
        last = now;
        if (mc.level == null || mc.player == null) return;
        float target = target(mc, partial);
        fade += (target - fade) * (1f - (float) Math.exp(-dt * 1.1f));
        if (Math.abs(target - fade) < 0.004f) fade = target;
        if (fade <= 0.01f) return;
        clock += dt;
        if (!Projector.prepare(mc, partial, w, h)) return;

        int lvl = Perf.level();
        boolean land = hills.on();
        if (land) skyline.sweep(mc);
        final boolean useLand = land && skyline.ready();
        AuroraSky.View view = new AuroraSky.View() {
            @Override
            public boolean at(float sx, float sy, float[] azel) {
                if (!Projector.ray(sx, sy, dir)) return false;
                azel[0] = (float) Math.toDegrees(Math.atan2(dir[0], -dir[2]));
                azel[1] = (float) Math.toDegrees(Math.asin(Math.max(-1.0, Math.min(1.0, dir[1]))));
                return true;
            }

            @Override
            public boolean hasSkyline() {
                return useLand;
            }

            @Override
            public float skyline(float az) {
                return skyline.at(az);
            }
        };
        try {
            int q = detail.index();
            if (lvl == 1) q = Math.max(0, q - 1);
            else if (lvl >= 2) q = 0;
            look.gain = fade * brightness.f() / 100f;
            look.motion = lvl >= 2 ? Math.min(0.6f, (float) motion.get() / 100f) : (float) motion.get() / 100f;
            look.twinkle = twinkle.on() ? 1f : 0f;
            look.cellsX = CELLS[q][0];
            look.cellsY = CELLS[q][1];
            look.cutLand = useLand;
            photo.render(g, w, h, view, look, clock);
            int gs = Math.max(1, (int) mc.getWindow().getGuiScale());
            scape.updateMeteors(clock, lvl >= 2 ? 0f : METEOR_GAP[shooting.index()]);
            scape.renderMeteors(g, view, fade, gs);
        } catch (RuntimeException | LinkageError e) {
            if (!warned) com.zlorbie788.zeus.core.Zeus.LOG.warn("The atmosphere could not be drawn", e);
            warned = true;
        }
    }

    /** How visible the sky should be right now, 0 to 1, and the reason in {@link #why}. */
    private float target(Minecraft mc, float partial) {
        if (mc.level.dimension() != Level.OVERWORLD) {
            why = "only in the overworld";
            return 0f;
        }
        float nightNow = 1f;
        if (when.index() != 2) {
            try {
                net.minecraft.world.phys.Vec3 at = mc.player != null ? mc.player.position() : net.minecraft.world.phys.Vec3.ZERO;
                float st = mc.level.environmentAttributes().getValue(net.minecraft.world.attribute.EnvironmentAttributes.STAR_BRIGHTNESS, at);
                nightNow = Math.max(0f, Math.min(1f, (st - 0.05f) / 0.28f));
            } catch (RuntimeException | LinkageError e) {
                nightNow = 1f;
            }
        }
        float rain = 0f;
        if (when.index() == 0) {
            try {
                rain = Math.max(mc.level.getRainLevel(partial), mc.level.getThunderLevel(partial));
            } catch (RuntimeException | LinkageError e) {
                rain = 0f;
            }
        }
        int skyLight = 15;
        try {
            BlockPos p = mc.player.blockPosition().above();
            skyLight = mc.level.getBrightness(LightLayer.SKY, p);
        } catch (RuntimeException | LinkageError e) {
            skyLight = 15;
        }
        float open = Math.max(0f, Math.min(1f, (skyLight - 5f) / 8f));
        if (nightNow < 0.01f) why = "it is day";
        else if (rain > 0.5f) why = "rain";
        else if (open < 0.05f) why = "no open sky";
        else why = when.index() == 2 ? "always on" : "clear night";
        return nightNow * (1f - rain) * open;
    }
}
