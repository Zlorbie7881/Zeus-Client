package com.zlorbie788.zeus.mods;

import com.zlorbie788.zeus.core.Category;
import com.zlorbie788.zeus.core.Module;
import com.zlorbie788.zeus.core.Setting;
import com.zlorbie788.zeus.render.Projector;
import com.zlorbie788.zeus.ui.AuroraSky;
import com.zlorbie788.zeus.ui.Perf;
import net.minecraft.client.Minecraft;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.core.BlockPos;
import net.minecraft.world.level.Level;
import net.minecraft.world.level.LightLayer;
import net.minecraft.world.level.levelgen.Heightmap;

import java.util.List;

/**
 * Northern lights in the night sky of your world. A rebuilt, much richer version of the idea: instead of three flat ribbons it
 * is a sky of nested, folding curtains with rays, a pink fringe along the lower edge, colours that change with height and
 * surges that wake the whole sky now and then (see {@link AuroraSky}).
 *
 * <p>It is drawn on top of the game picture, like the other world overlays, from the camera's own view, so it turns with you,
 * and it is hidden behind hills and trees: a ring of heights around you is measured a few bearings at a time and the aurora is
 * cut along that skyline. It fades in at night, out at dawn, in the rain, and when you are under a roof or in a cave.</p>
 */
public final class Aurora extends Module {
    private final Setting.Choice colours = add(new Setting.Choice("Colours", "The colours of the curtains from the bottom up.", 0, AuroraSky.SCHEMES));
    private final Setting.Num intensity = add(new Setting.Num("Brightness", "How bright the lights are.", 100, 20, 200, 10, "%"));
    private final Setting.Num curtains = add(new Setting.Num("Curtains", "How many arcs lie above each other.", 4, 1, 6, 1, ""));
    private final Setting.Num height = add(new Setting.Num("Height", "How high above the horizon the lowest arc starts.", 16, 4, 45, 1, " deg"));
    private final Setting.Choice direction = add(new Setting.Choice("Direction", "Where the arcs stand. All around fills the whole sky.", 0, "North", "East", "South", "West", "All around"));
    private final Setting.Num length = add(new Setting.Num("Arc length", "How far the arcs reach to each side.", 62, 30, 120, 2, " deg")).showIf(() -> direction.index() != 4);
    private final Setting.Num speed = add(new Setting.Num("Speed", "How fast the folds and rays move.", 100, 30, 250, 10, "%"));
    private final Setting.Bool surges = add(new Setting.Bool("Surges", "Every minute or two the sky wakes up: brighter arcs, rays shooting up, a red top and a corona overhead.", true));
    private final Setting.Choice when = add(new Setting.Choice("Show", "When the lights are visible.", 0, "Clear nights", "Nights, also in rain", "Always"));
    private final Setting.Bool hills = add(new Setting.Bool("Hide behind land", "The lights go behind hills, mountains and trees instead of floating in front of them.", true));
    private final Setting.Choice quality = add(new Setting.Choice("Quality", "Finer columns and smoother colour on higher settings. Zeus lowers it by itself when the frame rate is low.", 2, "Low", "Medium", "High", "Ultra"));
    private final Setting.Info status = add(new Setting.Info("Status", this::status));

    private static final int[] COLS = {90, 150, 220, 320};
    private static final int[] ROWS = {20, 30, 42, 58};

    private final AuroraSky sky = new AuroraSky();
    private final AuroraSky.Style style = new AuroraSky.Style();
    private final Skyline skyline = new Skyline();
    private final float[] ray = new float[3];
    private final double[] dir = new double[3];

    private float fade;
    private float clock;
    private long last;
    private float night;
    private float rainFade = 1f;
    private float open;
    private String why = "";
    private boolean warned;
    private int columnsNow;

    public Aurora() {
        super("aurora", "Aurora", "Northern lights in the night sky: folding green and violet curtains with rays and surges.", Category.RENDER);
        style.seed = 3 + (int) (System.nanoTime() & 0x3FF);
    }

    @Override
    protected void onDisable() {
        fade = 0f;
        last = 0L;
        skyline.reset();
    }

    private List<String> status() {
        if (!enabled()) return List.of("Switched off");
        return List.of(
                "Showing: " + (fade > 0.02f ? Math.round(fade * 100f) + "%" : "hidden"),
                "Why: " + (why.isEmpty() ? "-" : why),
                "Surge: " + Math.round(sky.surge() * 100f) + "%",
                "Skyline: " + (!hills.on() ? "off" : skyline.ready() ? "measured" : "measuring"));
    }

    @Override
    public String info() {
        return colours.value();
    }

    @Override
    protected void onHudWorld(GuiGraphics g, Minecraft mc, float partial, int w, int h) {
        long now = System.nanoTime();
        float dt = last == 0L ? 0f : Math.min(0.1f, (now - last) / 1e9f);
        last = now;
        if (mc.level == null || mc.player == null) return;
        float target = target(mc, partial);
        fade += (target - fade) * (1f - (float) Math.exp(-dt * 0.9f));
        if (Math.abs(target - fade) < 0.004f) fade = target;
        if (fade <= 0.01f) return;
        clock += dt * (float) speed.get() / 100f;
        if (!Projector.prepare(mc, partial, w, h)) return;

        style.curtains = curtains.i();
        style.scheme = colours.index();
        style.height = height.f();
        int dirIndex = direction.index();
        style.bearing = dirIndex == 4 ? 0f : dirIndex * 90f;
        style.spread = dirIndex == 4 ? 178f : length.f();
        style.activity = 0.78f;
        style.surges = surges.on();
        sky.begin(clock, style);

        int q = quality.index();
        int lvl = Perf.level();
        if (lvl >= 2) q = 0;
        else if (lvl == 1) q = Math.max(0, q - 1);
        int col = Math.max(2, Math.round(w / (float) COLS[q]));
        int row = Math.max(4, Math.round(h / (float) ROWS[q]));
        columnsNow = w / col;

        boolean land = hills.on();
        if (land) skyline.sweep(mc);
        final boolean useLand = land && skyline.ready();
        AuroraSky.View view = new AuroraSky.View() {
            @Override
            public boolean at(float sx, float sy, float[] azel) {
                if (!Projector.ray(sx, sy, dir)) return false;
                azel[0] = (float) Math.toDegrees(Math.atan2(dir[0], -dir[2]));
                azel[1] = (float) Math.toDegrees(Math.asin(Math.max(-1.0, Math.min(1.0, dir[1]))));
                return true;
            }

            @Override
            public boolean hasSkyline() {
                return useLand;
            }

            @Override
            public float skyline(float az) {
                return skyline.at(az);
            }
        };
        try {
            sky.render(g, 0, 0, w, h, view, fade * intensity.f() / 100f, col, row);
        } catch (RuntimeException | LinkageError e) {
            if (!warned) com.zlorbie788.zeus.core.Zeus.LOG.warn("The aurora could not be drawn", e);
            warned = true;
        }
    }

    /** How visible the lights should be right now, 0 to 1, and the reason in {@link #why}. */
    private float target(Minecraft mc, float partial) {
        if (mc.level.dimension() != Level.OVERWORLD) {
            why = "only in the overworld";
            return 0f;
        }
        float nightNow = 1f;
        if (when.index() != 2) {
            try {
                net.minecraft.world.phys.Vec3 at = mc.player != null ? mc.player.position() : net.minecraft.world.phys.Vec3.ZERO;
                float stars = mc.level.environmentAttributes().getValue(net.minecraft.world.attribute.EnvironmentAttributes.STAR_BRIGHTNESS, at);
                nightNow = Math.max(0f, Math.min(1f, (stars - 0.05f) / 0.28f));
            } catch (RuntimeException | LinkageError e) {
                nightNow = 1f;
            }
        }
        float rain = 0f;
        if (when.index() == 0) {
            try {
                rain = Math.max(mc.level.getRainLevel(partial), mc.level.getThunderLevel(partial));
            } catch (RuntimeException | LinkageError e) {
                rain = 0f;
            }
        }
        int sky = 15;
        try {
            BlockPos p = mc.player.blockPosition().above();
            sky = mc.level.getBrightness(LightLayer.SKY, p);
        } catch (RuntimeException | LinkageError e) {
            sky = 15;
        }
        float openNow = Math.max(0f, Math.min(1f, (sky - 5f) / 8f));
        if (nightNow < 0.01f) why = "it is day";
        else if (rain > 0.5f) why = "rain";
        else if (openNow < 0.05f) why = "no open sky";
        else why = when.index() == 2 ? "always on" : "clear night";
        return nightNow * (1f - rain) * openNow;
    }

    /** The heights of the land around you, as an elevation for each bearing, measured a few bearings every frame. */
    static final class Skyline {
        private static final int BINS = 240;
        private static final double[] DIST = {3, 5, 8, 12, 17, 24, 34, 48, 68, 96, 136, 190};
        private final float[] elev = new float[BINS];
        private boolean ready;
        private int cursor;
        private boolean sweeping;
        private double sx;
        private double sz;
        private long doneAt;
        private double eyeY;
        private double ex;
        private double ez;

        void reset() {
            ready = false;
            sweeping = false;
            cursor = 0;
            doneAt = 0L;
        }

        boolean ready() {
            return ready;
        }

        float at(float az) {
            float f = az / 360f * BINS;
            f %= BINS;
            if (f < 0f) f += BINS;
            int i0 = (int) f;
            int i1 = i0 + 1 == BINS ? 0 : i0 + 1;
            float fr = f - i0;
            return elev[i0] + (elev[i1] - elev[i0]) * fr;
        }

        void sweep(Minecraft mc) {
            if (mc.level == null || mc.player == null) return;
            long now = System.currentTimeMillis();
            double px = mc.player.getX();
            double pz = mc.player.getZ();
            if (!sweeping) {
                double moved = Math.abs(px - sx) + Math.abs(pz - sz);
                if (ready && moved < 2.0 && now - doneAt < 1500L) return;
                sweeping = true;
                cursor = 0;
                ex = px;
                ez = pz;
                eyeY = mc.player.getEyeY();
                sx = px;
                sz = pz;
            }
            int maxDist = 190;
            try {
                maxDist = Math.min(190, mc.options.getEffectiveRenderDistance() * 16);
            } catch (RuntimeException | LinkageError e) {
                maxDist = 128;
            }
            int perFrame = ready ? 30 : BINS;
            for (int n = 0; n < perFrame && cursor < BINS; n++, cursor++) {
                double az = Math.toRadians((cursor + 0.5) * 360.0 / BINS);
                double sin = Math.sin(az);
                double cos = Math.cos(az);
                double best = -90.0;
                for (double d : DIST) {
                    if (d > maxDist) break;
                    int x = (int) Math.floor(ex + sin * d);
                    int z = (int) Math.floor(ez - cos * d);
                    int top = mc.level.getHeight(Heightmap.Types.MOTION_BLOCKING, x, z);
                    if (top <= mc.level.getMinY() + 1) continue;
                    double e = Math.toDegrees(Math.atan2(top - eyeY, d));
                    if (e > best) best = e;
                }
                elev[cursor] = (float) best;
            }
            if (cursor >= BINS) {
                sweeping = false;
                ready = true;
                doneAt = now;
            }
        }
    }
}
